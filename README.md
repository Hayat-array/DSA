## ⚙️ GitHub Actions Setup

</div>

<details>
<summary><strong>🐍 Snake Animation Workflow (snake.yml)</strong></summary>

```yaml
name: Generate Snake
on:
  schedule:
    - cron: "0 */12 * * *"
  workflow_dispatch:
permissions:
  contents: write
jobs:
  generate:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      - name: Generate Snake
        uses: Platane/snk@v3
        with:
          github_user_name: Hayat-array
          outputs: |
            dist/github-contribution-grid-snake.svg
            dist/github-contribution-grid-snake-dark.svg?palette=github-dark
      - name: Deploy to output branch
        uses: crazy-max/ghaction-github-pages@v3
        with:
          target_branch: output
          build_dir: dist
          keep_history: true
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
```
</details>
<details> 
<summary><strong>📊 Auto Problem Counter Workflow (count-problems.yml)</strong></summary>
  
```yaml
name: Count DSA Problems
on:
  push:
    branches: [main]
  workflow_dispatch:
permissions:
  contents: write
jobs:
  count:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      - name: Count Problems
        run: |
          EASY=$(find . -name "easy_*" | wc -l)
          MEDIUM=$(find . -name "medium_*" | wc -l)
          HARD=$(find . -name "hard_*" | wc -l)
          echo "{\"easy\": $EASY, \"medium\": $MEDIUM, \"hard\": $HARD}" > stats.json
      - name: Commit Stats
        run: |
          git config --local user.email "action@github.com"
          git config --local user.name "GitHub Action"
          git add stats.json
          git diff --staged --quiet || git commit -m "📊 Auto-update problem stats"
          git push
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
```
</details>
